\node[ppl, w] (msg) at (4.6,0) {a request in plain words: portal, chat, email};

\node[slv, r] (neg) at (15.4,-10.4) {\textbf{negotiate}: MUS $\to$ every MCS, each re-solved $\to$ ranked};
\node[mdl, r] (ask) at (15.4,-8.0) {offer 2--3 \emph{verified} options + a grounded explanation};
\node[mdl, r] (rep) at (15.4,-5.6) {each reply $\to$ exactly one typed tool call};
\node[bad, r] (esc) at (15.4,-3.2) {no reply or deadlock $\to$ escalate to a person};
